<script>
(function(){
  var sec=document.getElementById('tabiFaq');if(!sec)return;
  // Reveal left panel
  var left=document.getElementById('tabiFaqLeft');
  if(left){
    var lo=new IntersectionObserver(function(en){if(!en[0].isIntersecting)return;left.classList.add('is-visible');lo.disconnect();},{rootMargin:'-60px 0px'});
    lo.observe(left);
  }
  // Reveal headline lines
  sec.querySelectorAll('.tabi-faq__hl-line').forEach(function(el){
    var o=new IntersectionObserver(function(en){if(!en[0].isIntersecting)return;el.classList.add('is-visible');o.disconnect();},{rootMargin:'-40px 0px'});
    o.observe(el);
  });
  // Reveal + stagger items
  sec.querySelectorAll('.tabi-faq__item').forEach(function(item,i){
    var o=new IntersectionObserver(function(en){
      if(!en[0].isIntersecting)return;
      setTimeout(function(){item.classList.add('is-visible');},i*60);
      o.disconnect();
    },{rootMargin:'-40px 0px'});
    o.observe(item);
  });
  // Accordion toggle
  sec.querySelectorAll('.tabi-faq__trigger').forEach(function(btn){
    btn.addEventListener('click',function(){
      var item=btn.closest('.tabi-faq__item');
      var isOpen=item.classList.contains('tabi-faq__item--open');
      // close all
      sec.querySelectorAll('.tabi-faq__item--open').forEach(function(el){el.classList.remove('tabi-faq__item--open');el.querySelector('.tabi-faq__trigger').setAttribute('aria-expanded','false');});
      if(!isOpen){item.classList.add('tabi-faq__item--open');btn.setAttribute('aria-expanded','true');}
    });
  });
})();
</script>
