<script>
(function(){
  var aboutEl=document.getElementById('tabiAbout');
  if(!aboutEl)return;
  function io(els,cb,margin){
    var obs=new IntersectionObserver(function(en){
      en.forEach(function(e){if(!e.isIntersecting)return;cb(e.target);obs.unobserve(e.target);});
    },{rootMargin:margin||'-60px 0px'});
    els.forEach(function(el){obs.observe(el);});
  }
  // Reveals
  io(aboutEl.querySelectorAll('.tabi-reveal'),function(el){el.classList.add('is-visible');});
  io(aboutEl.querySelectorAll('.tabi-hl-line'),function(el){el.classList.add('is-visible');},'-40px 0px');
  io(aboutEl.querySelectorAll('.tabi-pillar'),function(el){
    el.classList.add('is-visible');
    var f=el.querySelector('.tabi-pillar__bar-fill');if(f)f.classList.add('is-visible');
  },'-50px 0px');
  // Eyebrow dot
  var dot=document.getElementById('tabiAboutDot');
  if(dot){io([dot.closest('.tabi-about__eyebrow')||dot],function(){dot.classList.add('is-visible');});}
  // Stats counters
  aboutEl.querySelectorAll('.tabi-stat').forEach(function(s){
    var target=parseInt(s.dataset.target,10)||0;
    var delay=parseFloat(s.dataset.delay||'0');
    var fill=s.querySelector('.tabi-stat__bar-fill');
    var val=s.querySelector('.tabi-stat__val');
    var done=false;
    var obs=new IntersectionObserver(function(en){
      if(!en[0].isIntersecting||done)return;done=true;
      setTimeout(function(){
        if(fill)fill.classList.add('is-visible');
        if(!val)return;
        var t0=performance.now(),dur=1400;
        (function tick(now){var t=Math.min((now-t0)/dur,1);val.textContent=Math.round(target*(1-Math.pow(1-t,3)));if(t<1)requestAnimationFrame(tick);})(performance.now());
      },delay*1000);
      obs.disconnect();
    },{rootMargin:'-60px 0px'});
    obs.observe(s);
  });
  // Kanji parallax (needs GSAP)
  var kanji=document.getElementById('tabiAboutKanji');
  function initKanji(){
    if(!window.gsap||!window.ScrollTrigger)return;
    gsap.registerPlugin(ScrollTrigger);
    ScrollTrigger.create({trigger:aboutEl,start:'top bottom',end:'bottom top',onUpdate:function(s){
      var p=s.progress;
      kanji.style.transform='translateY('+((p*-18)+8)+'%) rotate('+((p*4)-2)+'deg)';
      kanji.style.opacity=p<.15?(p/.15)*.055:p>.7?((1-p)/.3)*.055:.055;
    }});
  }
  function loadST(cb){
    var G='https://cdnjs.cloudflare.com/ajax/libs/gsap/3.12.5/gsap.min.js';
    var ST='https://cdnjs.cloudflare.com/ajax/libs/gsap/3.12.5/ScrollTrigger.min.js';
    function ls(src,next){if(document.querySelector('script[src="'+src+'"]')){next&&next();return;}var s=document.createElement('script');s.src=src;s.onload=next||null;document.head.appendChild(s);}
    if(window.gsap&&window.ScrollTrigger){cb();}
    else if(window.gsap){ls(ST,cb);}
    else{ls(G,function(){ls(ST,cb);});}
  }
  if(kanji)loadST(initKanji);
})();
</script>
