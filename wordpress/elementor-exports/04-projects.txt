<script>
(function(){
  var sec=document.getElementById('tabiProj');if(!sec)return;
  function io(els,cb,margin){
    var obs=new IntersectionObserver(function(en){
      en.forEach(function(e){if(!e.isIntersecting)return;cb(e.target);obs.unobserve(e.target);});
    },{rootMargin:margin||'-60px 0px'});
    (Array.isArray(els)?els:Array.from(els)).forEach(function(el){obs.observe(el);});
  }
  var ey=document.getElementById('tabiProjEyebrow');
  if(ey)io([ey],function(el){el.classList.add('is-visible');});
  io(sec.querySelectorAll('.tabi-hl-line'),function(el){el.classList.add('is-visible');},'-40px 0px');
  io(sec.querySelectorAll('.tabi-proj__card'),function(card){card.classList.add('is-visible');},'-80px 0px');
  // Hover border accent
  sec.querySelectorAll('.tabi-proj__card:not(.tabi-proj__card--cta)').forEach(function(card){
    var dot=card.querySelector('.tabi-proj__dot');
    var border=card.querySelector('.tabi-proj__border');
    var accent=dot?dot.style.backgroundColor:'#F20C25';
    card.addEventListener('mouseenter',function(){if(border)border.style.borderColor=accent;});
    card.addEventListener('mouseleave',function(){if(border)border.style.borderColor='transparent';});
  });
})();
</script>
